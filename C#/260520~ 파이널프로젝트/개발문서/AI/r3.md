## 1. 개요

`R3`는 UniRx를 계승한 **최신 Reactive Extensions** 구현체다.  
`Observable<T>` 스트림으로 이벤트, 상태 변화, 비동기 흐름을 선언적으로 조합한다.

| UniRx | R3 |
|---|---|
| `Subject<T>` | `Subject<T>` (동일) |
| `ReactiveProperty<T>` | `ReactiveProperty<T>` (동일) |
| `Observable.Timer` | `Observable.Timer` |
| `IDisposable` 구독 | `IDisposable` 구독 (동일) |
| UniTask 연동 불완전 | UniTask 네이티브 연동 |

## 3. Observable 기초

```csharp
using R3;
using UnityEngine;

public class BasicExample : MonoBehaviour
{
    private void Start()
    {
        // ── 팩토리 메서드 ─────────────────────────────────
        Observable.Return(42)              // 단일 값 emit 후 완료
            .Subscribe(x => Debug.Log(x));

        Observable.Range(0, 5)             // 0~4 순차 emit
            .Subscribe(x => Debug.Log(x));

        Observable.Interval(TimeSpan.FromSeconds(1)) // 1초마다 emit
            .Subscribe(x => Debug.Log($"틱: {x}"));

        Observable.Timer(TimeSpan.FromSeconds(3))    // 3초 후 단발
            .Subscribe(_ => Debug.Log("3초 경과"));

        // ── Subscribe 오버로드 ────────────────────────────
        Observable.Range(1, 3)
            .Subscribe(
                onNext:      x => Debug.Log($"값: {x}"),
                onError:     e => Debug.LogError(e),
                onCompleted: () => Debug.Log("완료")
            );
    }
}
```

## 4. Subject 종류

```csharp
// ── Subject<T> : 일반 브로드캐스트 ───────────────────────
var subject = new Subject<int>();
subject.Subscribe(x => Debug.Log($"수신: {x}"));
subject.OnNext(1);   // 수신: 1
subject.OnNext(2);   // 수신: 2
subject.OnCompleted();

// ── BehaviorSubject<T> : 마지막 값 유지, 구독 즉시 방출 ──
var behavior = new BehaviorSubject<int>(0); // 초기값 0
behavior.Subscribe(x => Debug.Log($"BehaviorSubject: {x}")); // 즉시 0 출력
behavior.OnNext(10); // 10 출력

// ── ReplaySubject<T> : 지정 개수만큼 이전 값 재방출 ──────
var replay = new ReplaySubject<int>(bufferSize: 3);
replay.OnNext(1);
replay.OnNext(2);
replay.OnNext(3);
replay.Subscribe(x => Debug.Log($"Replay: {x}")); // 1, 2, 3 즉시 출력

// ── AsyncSubject<T> : OnCompleted 시 마지막 값 방출 ──────
var async_ = new AsyncSubject<int>();
async_.Subscribe(x => Debug.Log($"AsyncSubject: {x}"));
async_.OnNext(1);
async_.OnNext(2);
async_.OnCompleted(); // 마지막 값 2만 출력
```

## 5. ReactiveProperty

상태를 Observable로 래핑한다. 값 변경 시 자동으로 스트림이 흐른다.

```csharp
using R3;
using UnityEngine;

public class PlayerModel : MonoBehaviour
{
    // ── 선언 ─────────────────────────────────────────────
    public ReactiveProperty<int> Hp { get; } = new(100);
    public ReactiveProperty<bool> IsDead { get; } = new(false);

    // ── 읽기 전용 노출 ────────────────────────────────────
    private readonly ReactiveProperty<int> _score = new(0);
    public ReadOnlyReactiveProperty<int> Score => _score;

    private void Start()
    {
        // 값 변경 구독
        Hp.Subscribe(hp => Debug.Log($"HP: {hp}"));

        // 파생 스트림 — HP가 0 이하이면 IsDead = true
        Hp.Where(hp => hp <= 0)
          .Subscribe(_ => IsDead.Value = true);

        // 값 변경
        Hp.Value = 50;  // HP: 50
        Hp.Value = 0;   // HP: 0, IsDead → true
    }

    private void OnDestroy()
    {
        Hp.Dispose();
        IsDead.Dispose();
        _score.Dispose();
    }
}
```

## 6. 주요 오퍼레이터

```csharp
var source = new Subject<int>();

// ── 필터링 ───────────────────────────────────────────────
source.Where(x => x > 0);                  // 양수만 통과
source.Distinct();                          // 중복 제거
source.DistinctUntilChanged();              // 연속 중복 제거
source.Skip(3);                             // 앞 3개 무시
source.Take(5);                             // 앞 5개만 통과
source.TakeLast(3);                         // 마지막 3개
source.IgnoreElements();                    // 완료/에러만 통과

// ── 변환 ─────────────────────────────────────────────────
source.Select(x => x * 2);                 // 맵핑
source.SelectMany(x => Observable.Range(0, x)); // 평탄화
source.Scan(0, (acc, x) => acc + x);       // 누적 합산
source.Buffer(3);                           // 3개씩 묶어서 방출

// ── 시간 ─────────────────────────────────────────────────
source.Throttle(TimeSpan.FromMilliseconds(300));  // 마지막 값만 (디바운스)
source.Sample(TimeSpan.FromSeconds(1));            // 1초 간격 샘플링
source.Debounce(TimeSpan.FromMilliseconds(300));  // 입력 중단 후 방출
source.Delay(TimeSpan.FromSeconds(1));             // 1초 지연

// ── 결합 ─────────────────────────────────────────────────
var other = new Subject<string>();
source.CombineLatest(other, (a, b) => $"{a}-{b}");  // 둘 다 최신값 조합
source.Merge(Observable.Return(99));                  // 스트림 병합
source.Zip(other, (a, b) => (a, b));                  // 쌍으로 결합

// ── 에러 처리 ─────────────────────────────────────────────
source.Catch<int, Exception>(e =>
{
    Debug.LogError(e);
    return Observable.Empty<int>();
});
source.Retry(3);                            // 에러 시 최대 3회 재시도
source.OnErrorRetry((Exception e) => { }, retryCount: 3, delay: TimeSpan.FromSeconds(1));
```

## 7. Unity 전용 확장

```csharp
using R3;
using R3.Triggers;
using UnityEngine;
using UnityEngine.UI;

public class UnityExtensionExample : MonoBehaviour
{
    [SerializeField] private Button _button;
    [SerializeField] private Slider _slider;

    private void Start()
    {
        // ── UI 이벤트 ─────────────────────────────────────
        _button.OnClickAsObservable()
            .Subscribe(_ => Debug.Log("클릭"))
            .AddTo(this);

        _slider.OnValueChangedAsObservable()
            .DistinctUntilChanged()
            .Subscribe(v => Debug.Log($"슬라이더: {v}"))
            .AddTo(this);

        // ── MonoBehaviour 이벤트 스트림 ──────────────────
        this.OnUpdateAsObservable()
            .Subscribe(_ => { /* 매 Update */ })
            .AddTo(this);

        this.OnTriggerEnterAsObservable()
            .Subscribe(col => Debug.Log($"트리거: {col.name}"))
            .AddTo(this);

        this.OnCollisionEnterAsObservable()
            .Where(col => col.gameObject.CompareTag("Enemy"))
            .Subscribe(col => Debug.Log("적 충돌"))
            .AddTo(this);

        // ── Observable.EveryUpdate (전역) ────────────────
        Observable.EveryUpdate()
            .Take(100) // 100프레임만
            .Subscribe(_ => { })
            .AddTo(this);
    }
}
```

## 8. 구독 해제 (Dispose)

```csharp
public class DisposeExample : MonoBehaviour
{
    // 방법 1: CompositeDisposable — 여러 구독 한번에 해제
    private readonly CompositeDisposable _disposables = new();

    private void Start()
    {
        Observable.Interval(TimeSpan.FromSeconds(1))
            .Subscribe(x => Debug.Log(x))
            .AddTo(_disposables);  // CompositeDisposable에 추가

        // 방법 2: AddTo(MonoBehaviour) — 오브젝트 파괴 시 자동 해제 (권장)
        Observable.Interval(TimeSpan.FromSeconds(1))
            .Subscribe(x => Debug.Log(x))
            .AddTo(this);
    }

    private void OnDestroy()
    {
        _disposables.Dispose(); // 방법 1은 수동 해제 필요
    }
}
```

## 9. UniTask 연동

```csharp
using R3;
using Cysharp.Threading.Tasks;

// Observable → UniTask
var subject = new Subject<int>();
int value = await subject.FirstAsync();                   // 첫 번째 값 대기
int value2 = await subject.FirstOrDefaultAsync(x => x > 0); // 조건 첫 번째

// 완료 대기
await subject.LastAsync();

// Observable → IAsyncEnumerable
await foreach (var x in subject.ToAsyncEnumerable())
{
    Debug.Log(x);
}

// UniTask → Observable
Observable.FromAsync(ct => SomeUniTaskAsync(ct))
    .Subscribe(result => Debug.Log(result));
```

## 10. 주의사항 및 트러블슈팅

| 문제 | 원인 | 해결 |
|---|---|---|
| 구독이 해제되지 않아 메모리 누수 | `AddTo` 또는 `Dispose` 미사용 | `.AddTo(this)` 반드시 추가 |
| `OnCompleted` 후에도 스트림 흐름 | 완료된 Subject에 OnNext 호출 | Subject 상태 확인 후 OnNext |
| `DistinctUntilChanged` 미동작 | 참조형 동등성 비교 실패 | `IEqualityComparer<T>` 직접 전달 |
| UI 스레드 아닌 곳에서 UI 업데이트 | 백그라운드 스트림 | `.ObserveOnMainThread()` 추가 |
| Hot/Cold Observable 혼동 | Cold는 구독마다 새 스트림 | `Publish().RefCount()` 로 Hot 변환 |
